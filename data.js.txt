// Función utilitaria para interpolar precios si falta algún año intermedio
function obtenerPrecioBase(preciosObj, anio) {
  if (preciosObj[anio]) return preciosObj[anio];

  const aniosDisponibles = Object.keys(preciosObj).map(Number).sort((a, b) => b - a);
  const masReciente = aniosDisponibles[0];
  const masAntiguo = aniosDisponibles[aniosDisponibles.length - 1];

  if (anio > masReciente) {
    return Math.round(preciosObj[masReciente] * Math.pow(1.08, anio - masReciente));
  } else if (anio < masAntiguo) {
    return Math.round(preciosObj[masAntiguo] * Math.pow(0.91, masAntiguo - anio));
  } else {
    const anioSuperior = aniosDisponibles.find(a => a > anio);
    const anioInferior = [...aniosDisponibles].reverse().find(a => a < anio);
    const ratio = (anio - anioInferior) / (anioSuperior - anioInferior);
    return Math.round(preciosObj[anioInferior] + ratio * (preciosObj[anioSuperior] - preciosObj[anioInferior]));
  }
}
